// --- Utility Functions ---
const formatCurrency = (amount) => {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
  }).format(amount);
};

const formatDate = (dateString) => {
  const options = { month: 'short', day: 'numeric', year: 'numeric' };
  return new Date(dateString).toLocaleDateString('en-US', options);
};

const getRelativeDate = (daysAgo) => {
  const date = new Date();
  date.setDate(date.getDate() - daysAgo);
  return date.toISOString().split('T')[0];
};

const getIcon = (iconName) => {

  return IconComponent;
};

// Generate 30 days of somewhat realistic data
const generateMockExpenses = () => {
  const expenses = [];
  let idCounter = 1;
  
  // Recurring
  expenses.push({ id: `e${idCounter++}`, amount: 1200, categoryId: 'c1', date: getRelativeDate(2), notes: 'Rent', isRecurring: true });
  expenses.push({ id: `e${idCounter++}`, amount: 120, categoryId: 'c4', date: getRelativeDate(5), notes: 'Electric Bill', isRecurring: true });
  expenses.push({ id: `e${idCounter++}`, amount: 60, categoryId: 'c4', date: getRelativeDate(10), notes: 'Internet', isRecurring: true });
  expenses.push({ id: `e${idCounter++}`, amount: 15, categoryId: 'c5', date: getRelativeDate(15), notes: 'Netflix', isRecurring: true });

  // Variable daily/weekly
  const patterns = [
    { cat: 'c8', min: 3, max: 7, freq: 0.8, notes: ['Morning Coffee', 'Starbucks', 'Cafe'] }, // High freq coffee
    { cat: 'c2', min: 15, max: 60, freq: 0.5, notes: ['Lunch', 'Dinner out', 'Groceries', 'Snacks'] }, // Frequent food
    { cat: 'c3', min: 20, max: 50, freq: 0.3, notes: ['Gas', 'Uber', 'Train Ticket'] },
    { cat: 'c6', min: 40, max: 150, freq: 0.2, notes: ['Clothes', 'Amazon', 'Gadget'] },
    { cat: 'c5', min: 20, max: 100, freq: 0.15, notes: ['Movie', 'Concert', 'Game'] }
  ];

  for (let i = 0; i < 30; i++) {
    const currentDate = getRelativeDate(i);
    patterns.forEach(pattern => {
      if (Math.random() < pattern.freq) {
        expenses.push({
          id: `e${idCounter++}`,
          amount: Math.floor(Math.random() * (pattern.max - pattern.min + 1) + pattern.min),
          categoryId: pattern.cat,
          date: currentDate,
          notes: pattern.notes[Math.floor(Math.random() * pattern.notes.length)],
          isRecurring: false
        });
      }
    });
  }
  
  // Add a recent spike to trigger alerts
  expenses.push({ id: `e${idCounter++}`, amount: 180, categoryId: 'c2', date: getRelativeDate(1), notes: 'Fancy Dinner (Spike)', isRecurring: false });
  expenses.push({ id: `e${idCounter++}`, amount: 250, categoryId: 'c6', date: getRelativeDate(2), notes: 'Impulse Buy', isRecurring: false });


  return expenses.sort((a, b) => new Date(b.date) - new Date(a.date));
};

const INITIAL_EXPENSES = generateMockExpenses();

// --- Main Application Component ---
export default function BudgetTrackerApp() {
  const [currentView, setCurrentView] = useState('dashboard');
  const [expenses, setExpenses] = useState(INITIAL_EXPENSES);
  const [categories, setCategories] = useState(INITIAL_CATEGORIES);
  const [budgets, setBudgets] = useState(INITIAL_BUDGETS);

  // --- Derived Data & Analytics ---
  const totalSpent = useMemo(() => expenses.reduce((sum, e) => sum + e.amount, 0), [expenses]);
  
  const totalBudget = useMemo(() => budgets.reduce((sum, b) => sum + b.limit, 0), [budgets]);
  
  const categorySpending = useMemo(() => {
    const spending = {};
    expenses.forEach(e => {
      spending[e.categoryId] = (spending[e.categoryId] || 0) + e.amount;
    });
    return spending;
  }, [expenses]);

  const recentExpenses = useMemo(() => expenses.slice(0, 5), [expenses]);

  // --- Spending Behavior Detector ---
  const alerts = useMemo(() => {
    const newAlerts = [];
    const now = new Date();
    const oneWeekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
    const twoWeeksAgo = new Date(now.getTime() - 14 * 24 * 60 * 60 * 1000);

    // 1. Budget Exhaustion Alerts
    budgets.forEach(budget => {
      const spent = categorySpending[budget.categoryId] || 0;
      const percentage = (spent / budget.limit) * 100;
      const category = categories.find(c => c.id === budget.categoryId);
      
      if (percentage >= 90 && percentage < 100) {
        newAlerts.push({ id: `alert-warn-${budget.categoryId}`, type: 'warning', message: `Your ${category?.name} budget is nearly exhausted (${Math.round(percentage)}%).` });
      } else if (percentage >= 100) {
        newAlerts.push({ id: `alert-danger-${budget.categoryId}`, type: 'danger', message: `You have exceeded your ${category?.name} budget by ${formatCurrency(spent - budget.limit)}.` });
      }
    });

    // 2. Spending Spike Detection (This week vs Last week)
    categories.forEach(category => {
      const thisWeekSpent = expenses.filter(e => e.categoryId === category.id && new Date(e.date) >= oneWeekAgo).reduce((sum, e) => sum + e.amount, 0);
      const lastWeekSpent = expenses.filter(e => e.categoryId === category.id && new Date(e.date) >= twoWeeksAgo && new Date(e.date) < oneWeekAgo).reduce((sum, e) => sum + e.amount, 0);

      if (lastWeekSpent > 0) {
        const increase = ((thisWeekSpent - lastWeekSpent) / lastWeekSpent) * 100;
        if (increase > 50 && thisWeekSpent > 50) { // arbitrary thresholds for meaningful alerts
          newAlerts.push({ id: `alert-spike-${category.id}`, type: 'info', message: `${category.name} spending is up ${Math.round(increase)}% compared to last week.` });
        }
      }
    });

    return newAlerts;
  }, [expenses, budgets, categories, categorySpending]);


  // --- Actions ---
  const addExpense = (newExpense) => {
    setExpenses([{ ...newExpense, id: `e${Date.now()}` }, ...expenses]);
    setCurrentView('expenses');
  };

  const addCategory = (newCategory) => {
    setCategories([...categories, { ...newCategory, id: `c${Date.now()}` }]);
  };

  const updateBudget = (categoryId, newLimit) => {
    setBudgets(prev => {
      const existing = prev.find(b => b.categoryId === categoryId);
      if (existing) {
        return prev.map(b => b.categoryId === categoryId ? { ...b, limit: newLimit } : b);
      }
      return [...prev, { categoryId, limit: newLimit }];
    });
  };

  // --- Layout Wrappers ---
  const NavItem = ({ id, label, icon: Icon }) => (
    <button
      onClick={() => setCurrentView(id)}
      className={`w-full flex items-center space-x-3 px-4 py-3 rounded-xl transition-all duration-200 ${
        currentView === id
          ? 'bg-indigo-600 text-white shadow-md shadow-indigo-200/50'
          : 'text-slate-500 hover:bg-slate-100 hover:text-indigo-600'
      }`}
    >
      <Icon size={20} className={currentView === id ? 'text-white' : ''} />
      <span className="font-medium">{label}</span>
    </button>
  );

  return (
    <div className="flex h-screen bg-slate-50 font-sans overflow-hidden">
      {/* Sidebar - Desktop */}
      <aside className="w-64 bg-white border-r border-slate-200 flex flex-col hidden md:flex">
        <div className="p-6 flex items-center space-x-3">
          <div className="bg-indigo-600 p-2 rounded-lg">
            <Wallet className="text-white" size={24} />
          </div>
          <h1 className="text-xl font-bold text-slate-800 tracking-tight">FinTrack</h1>
        </div>
        <nav className="flex-1 px-4 space-y-1 overflow-y-auto">
          <NavItem id="dashboard" label="Dashboard" icon={LayoutDashboard} />
          <NavItem id="expenses" label="Expenses" icon={Receipt} />
          <NavItem id="add" label="Add Expense" icon={PlusCircle} />
          <NavItem id="budgets" label="Budgets" icon={Target} />
          <NavItem id="categories" label="Categories" icon={Tags} />
          <NavItem id="insights" label="Insights" icon={PieChartIcon} />
        </nav>
        <div className="p-4 border-t border-slate-200">
          <div className="bg-indigo-50 rounded-xl p-4 flex flex-col items-center text-center">
            <ShieldCheck className="text-indigo-600 mb-2" size={24} />
            <p className="text-xs text-slate-600 font-medium">Your data is secure</p>
          </div>
        </div>
      </aside>

      {/* Mobile Bottom Nav */}
      <div className="md:hidden fixed bottom-0 left-0 right-0 bg-white border-t border-slate-200 z-50 flex justify-around p-2 pb-safe">
        {[
          { id: 'dashboard', icon: LayoutDashboard },
          { id: 'expenses', icon: Receipt },
          { id: 'add', icon: PlusCircle },
          { id: 'budgets', icon: Target },
          { id: 'insights', icon: PieChartIcon },
        ].map(item => (
          <button
            key={item.id}
            onClick={() => setCurrentView(item.id)}
            className={`p-3 rounded-xl transition-colors ${currentView === item.id ? 'bg-indigo-100 text-indigo-600' : 'text-slate-400 hover:text-indigo-500'}`}
          >
            <item.icon size={24} />
          </button>
        ))}
      </div>

      {/* Main Content Area */}
      <main className="flex-1 overflow-y-auto pb-24 md:pb-0 scroll-smooth">
        <div className="max-w-6xl mx-auto p-4 md:p-8">
          
          {/* Header Mobile Only */}
          <div className="md:hidden flex items-center space-x-2 mb-6">
             <div className="bg-indigo-600 p-2 rounded-lg">
                <Wallet className="text-white" size={20} />
              </div>
              <h1 className="text-xl font-bold text-slate-800 tracking-tight">FinTrack</h1>
          </div>

          {currentView === 'dashboard' && (
            <DashboardView 
              totalSpent={totalSpent} 
              totalBudget={totalBudget} 
              recentExpenses={recentExpenses} 
              categories={categories}
              alerts={alerts}
              setCurrentView={setCurrentView}
            />
          )}
          {currentView === 'expenses' && (
            <ExpensesView 
              expenses={expenses} 
              categories={categories} 
            />
          )}
          {currentView === 'add' && (
            <AddExpenseView 
              categories={categories} 
              onSave={addExpense} 
              onCancel={() => setCurrentView('dashboard')} 
            />
          )}
          {currentView === 'categories' && (
            <CategoriesView 
              categories={categories} 
              onAddCategory={addCategory} 
            />
          )}
          {currentView === 'budgets' && (
            <BudgetsView 
              budgets={budgets} 
              categories={categories} 
              categorySpending={categorySpending}
              onUpdateBudget={updateBudget}
            />
          )}
          {currentView === 'insights' && (
            <InsightsView 
              expenses={expenses} 
              categories={categories}
              budgets={budgets}
              categorySpending={categorySpending}
            />
          )}
        </div>
      </main>
    </div>
  );
}

function BudgetsView({ budgets, categories, categorySpending, onUpdateBudget }) {
  return (
    <div className="space-y-6 animate-in fade-in duration-300">
       <header className="mb-8">
        <h2 className="text-3xl font-bold text-slate-800">Budgets</h2>
        <p className="text-slate-500 mt-1">Keep your spending in check.</p>
      </header>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        {budgets.map(budget => {
          const category = categories.find(c => c.id === budget.categoryId);
          if(!category) return null;
          
          const spent = categorySpending[budget.categoryId] || 0;
          const percentage = Math.min((spent / budget.limit) * 100, 100);
          const isOver = spent > budget.limit;
          const Icon = getIcon(category.iconName);

          // Color logic based on threshold
          let progressColor = 'bg-indigo-500';
          if (percentage > 90) progressColor = 'bg-red-500';
          else if (percentage > 75) progressColor = 'bg-amber-400';

          return (
            <div key={budget.categoryId} className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100">
              <div className="flex justify-between items-center mb-4">
                <div className="flex items-center space-x-3">
                   <div className="w-10 h-10 rounded-xl flex items-center justify-center text-white" style={{ backgroundColor: category.color }}>
                    <Icon size={18} />
                  </div>
                  <div>
                    <h3 className="font-bold text-slate-800">{category.name}</h3>
                    <p className="text-xs text-slate-500 font-medium uppercase tracking-wider">{category.group}</p>
                  </div>
                </div>
                <div className="text-right">
                  <p className={`font-bold ${isOver ? 'text-red-500' : 'text-slate-800'}`}>
                    {formatCurrency(spent)}
                  </p>
                  <p className="text-xs text-slate-400 font-medium">of {formatCurrency(budget.limit)}</p>
                </div>
              </div>
              
              <div className="relative pt-2">
                <div className="overflow-hidden h-3 mb-2 text-xs flex rounded-full bg-slate-100">
                  <div style={{ width: `${percentage}%` }} className={`shadow-none flex flex-col text-center whitespace-nowrap text-white justify-center transition-all duration-500 ${progressColor}`}></div>
                </div>
                <div className="flex justify-between text-xs font-semibold text-slate-400">
                  <span>{Math.round(percentage)}%</span>
                  {isOver && <span className="text-red-500 flex items-center"><AlertCircle size={12} className="mr-1"/> Over budget</span>}
                  {!isOver && <span>{formatCurrency(budget.limit - spent)} left</span>}
                </div>
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}

function InsightsView({ expenses, categories, categorySpending, budgets }) {
  // --- Prepare Data for Charts ---
  
  // Pie Chart Data
  const pieData = Object.entries(categorySpending).map(([catId, amount]) => {
    const cat = categories.find(c => c.id === catId);
    return { name: cat?.name || 'Unknown', value: amount, color: cat?.color || '#ccc' };
  }).filter(d => d.value > 0).sort((a,b) => b.value - a.value);

  // Grouped Spending (Needs vs Wants)
  const groupedSpending = categories.reduce((acc, cat) => {
    acc[cat.group] = (acc[cat.group] || 0) + (categorySpending[cat.id] || 0);
    return acc;
  }, {});
  const totalSpent = Object.values(categorySpending).reduce((sum, v) => sum + v, 0);
  const wantsRatio = (groupedSpending['Wants'] || 0) / totalSpent;

  // Trend Data (Last 7 days)
  const trendData = useMemo(() => {
    const days = [];
    for(let i=6; i>=0; i--) {
      const d = getRelativeDate(i);
      const dailyTotal = expenses.filter(e => e.date === d).reduce((sum, e) => sum + e.amount, 0);
      days.push({ name: formatDate(d).split(' ')[0] + ' ' + formatDate(d).split(' ')[1], total: dailyTotal });
    }
    return days;
  }, [expenses]);


  // --- Smart Saving Plan Logic ---
  const savingPlan = useMemo(() => {
    // Find top 'Wants' spending
    const wantsCategories = categories.filter(c => c.group === 'Wants');
    const wantsSpend = wantsCategories.map(c => ({ id: c.id, name: c.name, spent: categorySpending[c.id] || 0, iconName: c.iconName, color: c.color }))
                                      .sort((a,b) => b.spent - a.spent);
    
    const targets = wantsSpend.slice(0, 2).filter(c => c.spent > 20); // Top 2 wants to reduce
    const reductionRate = 0.20; // 20% reduction target
    
    let monthlySavings = 0;
    const actions = targets.map(target => {
      const saving = target.spent * reductionRate;
      monthlySavings += saving;
      return {
        ...target,
        saving,
        action: `Reduce ${target.name} by 20% (Cap at ${formatCurrency(target.spent - saving)})`
      }
    });

    return { actions, monthlySavings, yearlySavings: monthlySavings * 12 };
  }, [categories, categorySpending]);


  const CustomTooltip = ({ active, payload, label }) => {
    if (active && payload && payload.length) {
      return (
        <div className="bg-white p-3 shadow-lg rounded-xl border border-slate-100">
          <p className="font-semibold text-slate-800">{label || payload[0].name}</p>
          <p className="text-indigo-600 font-bold">{formatCurrency(payload[0].value)}</p>
        </div>
      );
    }
    return null;
  };

  return (
    <div className="space-y-8 animate-in fade-in duration-300">
       <header>
        <h2 className="text-3xl font-bold text-slate-800">Financial Insights</h2>
        <p className="text-slate-500 mt-1">Analyze your habits and optimize savings.</p>
      </header>

      {/* Top Level Metrics */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
         <div className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100 flex flex-col justify-center">
            <h3 className="text-sm font-semibold text-slate-500 uppercase tracking-wider mb-4">Needs vs Wants</h3>
            <div className="flex items-end space-x-2 mb-2">
              <span className="text-4xl font-black text-slate-800">{Math.round((groupedSpending['Needs']||0)/totalSpent*100 || 0)}%</span>
              <span className="text-slate-500 mb-1">Needs</span>
            </div>
            <div className="w-full flex h-4 rounded-full overflow-hidden mb-2">
                <div style={{width: `${(groupedSpending['Needs']||0)/totalSpent*100}%`}} className="bg-blue-500"></div>
                <div style={{width: `${(groupedSpending['Wants']||0)/totalSpent*100}%`}} className="bg-purple-400"></div>
            </div>
            <div className="flex justify-between text-xs font-medium text-slate-500">
                <span>Ideal: 50% Needs</span>
                <span>Ideal: 30% Wants</span>
            </div>
         </div>

         {/* Smart Saving Plan Widget */}
         <div className="bg-gradient-to-br from-emerald-400 to-teal-500 rounded-3xl p-6 shadow-md text-white">
            <div className="flex items-center space-x-2 mb-4">
              <Lightbulb className="text-emerald-100" />
              <h3 className="font-bold text-lg">Smart Saving Plan</h3>
            </div>
            <p className="text-emerald-50 text-sm mb-4">Based on your recent habits, here's your optimization plan:</p>
            
            <div className="space-y-3 mb-6">
              {savingPlan.actions.length > 0 ? savingPlan.actions.map(action => (
                <div key={action.id} className="bg-black/10 rounded-xl p-3 flex justify-between items-center backdrop-blur-sm">
                  <div className="flex items-center space-x-2">
                    <div className="w-2 h-2 rounded-full" style={{backgroundColor: action.color}}></div>
                    <span className="text-sm font-medium">{action.action}</span>
                  </div>
                  <span className="font-bold text-emerald-100">+{formatCurrency(action.saving)}</span>
                </div>
              )) : (
                <div className="text-sm bg-black/10 p-3 rounded-xl">Your spending is heavily optimized. Great job!</div>
              )}
            </div>

            <div className="border-t border-emerald-300/30 pt-4 flex justify-between items-end">
               <div>
                 <p className="text-xs text-emerald-100 font-medium uppercase">Potential Yearly Savings</p>
                 <p className="text-3xl font-black">{formatCurrency(savingPlan.yearlySavings)}</p>
               </div>
               <div className="bg-white/20 p-2 rounded-lg backdrop-blur-sm">
                 <TrendingUp size={20} />
               </div>
            </div>
         </div>
      </div>


      {/* Charts Section */}
      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
        
        {/* Pie Chart */}
        <div className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100 min-h-[400px] flex flex-col">
          <h3 className="text-lg font-bold text-slate-800 mb-6">Spending Breakdown</h3>
          <div className="flex-1 min-h-[300px]">
            <ResponsiveContainer width="100%" height="100%">
              <PieChart>
                <Pie
                  data={pieData}
                  cx="50%"
                  cy="50%"
                  innerRadius={80}
                  outerRadius={120}
                  paddingAngle={5}
                  dataKey="value"
                  stroke="none"
                >
                  {pieData.map((entry, index) => (
                    <Cell key={`cell-${index}`} fill={entry.color} />
                  ))}
                </Pie>
                <Tooltip content={<CustomTooltip />} />
                <Legend layout="horizontal" verticalAlign="bottom" align="center" iconType="circle" wrapperStyle={{ fontSize: '12px', paddingTop: '20px' }}/>
              </PieChart>
            </ResponsiveContainer>
          </div>
        </div>

        {/* Trend Line Chart */}
        <div className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100 min-h-[400px] flex flex-col">
          <h3 className="text-lg font-bold text-slate-800 mb-6">7-Day Trend</h3>
           <div className="flex-1 min-h-[300px]">
            <ResponsiveContainer width="100%" height="100%">
